    let cachedClientIp = '';
    async function getClientIp() {
      if (cachedClientIp) return cachedClientIp;
      try {
        const res = await fetch('https://api.ipify.org?format=json', { signal: AbortSignal.timeout(3000) });
        const data = await res.json();
        if (data && data.ip) {
          cachedClientIp = data.ip;
          return data.ip;
        }
      } catch (e) {
        try {
          const res2 = await fetch('https://api64.ipify.org?format=json', { signal: AbortSignal.timeout(2000) });
          const data2 = await res2.json();
          if (data2 && data2.ip) {
            cachedClientIp = data2.ip;
            return data2.ip;
          }
        } catch (e2) { }
      }
      return cachedClientIp || 'Unknown';
    }

    function formatCode(c) {
      if (!c) return '';
      var s = c.toString().trim();
      if (/^\d{4}$/.test(s)) return s;
      var digits = s.replace(/\D/g, '');
      if (digits.length >= 4) return digits.slice(-4);
      if (digits.length > 0) return digits.padStart(4, '0');
      return s;
    }

    function renderRoomsInPanel(gn, panel) {
      if (!gData) return;
      if (gData.dayClosed) { panel.innerHTML = '<div class="dayclosed-msg">🚫 ปิดให้บริการวันนี้</div>'; return; }
      const roomHiddenMap = gData.roomHidden || {};
      const rooms = (gData.groups[gn] || []); // ✅ แสดงทุกห้อง
      const now = new Date(), isToday = (currentDate === getLocalDateStr(now));
      const curMin = now.getHours() * 60 + now.getMinutes();
      panel.innerHTML = rooms.map(r => {
        const isClosed = (gData.roomStatus || {})[r] === false;
        const isHidden = !!roomHiddenMap[r];
        const slots = gData.timeSlots || [], tableData = (gData.tableData || {})[r] || {};
        let freeCount = 0;
        if (!isClosed && !isHidden) slots.slice(0, -1).forEach(t => {
          const [h, m] = t.split(':').map(Number);
          if (isToday && (h * 60 + m) < curMin) return;
          const s = tableData[t];
          if (!(s && s.status === 'busy' && !['เช็คเอาต์แล้ว', 'ออกแล้ว', 'ยกเลิกแล้ว'].includes(s.checkin))) freeCount++;
        });
        const badge = isClosed
          ? `<span style="font-size:0.72rem;color:#dc3545;font-weight:500;">ปิด</span>`
          : isHidden
            ? `<span style="font-size:0.72rem;color:#888;font-weight:500;">จองเคาน์เตอร์เท่านั้น</span>`
            : freeCount === 0
              ? `<span style="font-size:0.72rem;color:#999;font-weight:500;">เต็มแล้ว</span>`
              : `<span style="font-size:0.72rem;color:#1a9e5c;font-weight:600;">จองออนไลน์</span>`;
        const clickable = !isClosed && (freeCount > 0 || isHidden); // hidden ยังกดดูตารางได้
        return `<div class="room-card ${clickable ? 'room-card-clickable' : ''}"
      onclick="showSlots('${r}','${gn}')" style="margin:0;${isHidden ? 'opacity:0.7;cursor:pointer;' : clickable ? 'cursor:pointer;' : 'opacity:0.6;'}">
      <div class="room-card-header">
        <span class="room-name" style="${isHidden ? 'color:#aaa;' : ''}">${r}</span>
        <div style="display:flex;align-items:center;gap:8px;">
          ${badge}
          ${clickable ? '<span style="font-size:1rem;color:#bbb;">›</span>' : ''}
        </div>
      </div>
    </div>`;
      }).join('');
    }

    function buildStartTimes(room, dateStr, selectedStart, lockStart) {
      const sel = document.getElementById('m_start');
      if (!gData || gData.dayClosed) {
        sel.innerHTML = '<option>ปิดให้บริการ</option>'; sel.disabled = true;
        document.getElementById('m_end').innerText = 'ปิดให้บริการ'; return;
      }
      const now = new Date(), isToday = (dateStr === getLocalDateStr(now));
      const curMin = now.getHours() * 60 + now.getMinutes();
      const slots = gData.timeSlots || [], tableData = (gData.tableData || {})[room] || {};
      const opts = [];
      slots.slice(0, -1).forEach(t => {
        const [h, m] = t.split(':').map(Number);
        if (isToday && (h * 60 + m) < curMin) return;
        const s = tableData[t];
        const isBusy = s && s.status === 'busy' && !['เช็คเอาต์แล้ว', 'ออกแล้ว', 'ยกเลิกแล้ว'].includes(s.checkin);
        if (!isBusy) opts.push(t);
      });
      if (!opts.length) { sel.innerHTML = '<option>ไม่มีเวลาว่าง</option>'; sel.disabled = true; document.getElementById('m_end').innerText = '-'; return; }
      sel.disabled = lockStart;
      sel.innerHTML = opts.map(t => `<option value="${t}" ${t === selectedStart ? 'selected' : ''}>${t}</option>`).join('');
      updateEndTime(sel.value);
    }

    function updateEndTime(startTime) {
      const endSel = document.getElementById('m_end');
      const slots = gData && gData.timeSlots || [];
      const room = document.getElementById('m_room') && document.getElementById('m_room').value;
      const tableData = (gData && gData.tableData && room && gData.tableData[room]) || {};
      const idx = slots.indexOf(startTime);
      if (idx < 0 || idx >= slots.length - 1) {
        endSel.innerHTML = '<option>-</option>'; return;
      }
      // หา max slots — หยุดถ้าชน busy หรือครบ 8 (2 ชั่วโมง)
      // ✅ FIX: เดิมเช็ค slots[idx+i] (ช่องถัดจากตัวเลือกที่ i ไปอีก 1 ช่อง — คือช่อง "ขอบ" ที่ไม่ได้เป็นส่วนหนึ่งของการจองเลย)
      // ทำให้เวลาสิ้นสุดสูงสุดถูกตัดสั้นไป 1 ช่องเสมอ เช่น ว่าง 11:30, 11:45 แต่ 12:00 มีคนจอง — ควรเลือกจบได้ถึง 12:00
      // (ครอบคลุม 2 ช่องที่ว่างพอดี) แต่ของเดิมกลับไปเช็คว่า 12:00 ว่างไหม (ซึ่งไม่ว่าง) เลยจำกัดให้จบได้แค่ 11:45
      // ต้องเช็ค slots[idx+i-1] (ช่องที่ถูกเพิ่มเข้ามาจริงๆ ณ ตัวเลือกที่ i) แทน
      let maxSlots = 0;
      for (let i = 1; i <= 8; i++) {
        if (idx + i >= slots.length) break;
        const s = tableData[slots[idx + i - 1]];
        const busy = s && s.status === 'busy' && !['เช็คเอาต์แล้ว', 'ออกแล้ว', 'ยกเลิกแล้ว'].includes(s.checkin);
        if (busy) break;
        maxSlots = i;
      }
      const finalMax = Math.max(maxSlots, 1);

      endSel.innerHTML = '';
      for (let i = 1; i <= finalMax; i++) {
        const t = slots[idx + i];
        if (!t) break;
        const opt = document.createElement('option');
        opt.value = t;
        opt.text = t + ' น.';
        if (i === finalMax) opt.selected = true;
        endSel.appendChild(opt);
      }
      // ไม่ restore prevVal — default เป็น max เสมอ
    }

    function checkStudentIdsValidity(isSubmit = false) {
      const grid = document.getElementById('m_student_grid');
      if (!grid) return true;
      const capacity = parseInt(grid.dataset.capacity || '1');
      const inputs = Array.from(grid.querySelectorAll('input'));
      const errEl = document.getElementById('studentIdError');

      let hasEmpty = false;
      let hasInvalidFormat = false;

      const currentYearShort = (new Date().getFullYear() + 543) % 100;
      const minYear = currentYearShort - 8;

      // 1. ตรวจสอบรูปแบบทีละคน
      inputs.forEach(inp => {
        const val = inp.value.trim();
        if (val === '') {
          if (isSubmit) {
            inp.classList.add('input-error');
            hasEmpty = true;
          } else {
            inp.classList.remove('input-error');
          }
          return;
        }

        const idPrefix = parseInt(val.substring(0, 2), 10);
        const isCorrectLength = /^\d{10}$/.test(val);
        const isEngChula = val.endsWith("21");
        const isValidYear = (idPrefix >= minYear && idPrefix <= currentYearShort);

        if (!isCorrectLength || !isEngChula || !isValidYear) {
          inp.classList.add('input-error');
          hasInvalidFormat = true;
        } else {
          inp.classList.remove('input-error');
        }
      });

      // 2. ตรวจสอบรหัสซ้ำ
      const ids = inputs.map(i => i.value.trim()).filter(Boolean);
      const duplicates = ids.filter((item, index) => ids.indexOf(item) !== index);
      if (duplicates.length > 0) {
        inputs.forEach(inp => {
          if (duplicates.includes(inp.value.trim())) {
            inp.classList.add('input-error');
          }
        });
      }

      // แสดงข้อความตามลักษณะข้อผิดพลาด
      if (hasEmpty) {
        if (errEl) {
          errEl.innerText = `⚠️ กรุณากรอกรหัสนิสิตให้ครบ ${capacity} คน`;
          errEl.style.display = '';
        }
        return false;
      }
      if (hasInvalidFormat) {
        if (errEl) {
          errEl.innerText = '⚠️ ต้องเป็นรหัสของนิสิตคณะวิศวะเท่านั้น';
          errEl.style.display = '';
        }
        return false;
      }
      if (duplicates.length > 0) {
        if (errEl) {
          errEl.innerText = '⚠️ พบรหัสนิสิตซ้ำกันในกลุ่ม';
          errEl.style.display = '';
        }
        return false;
      }

      if (errEl) errEl.style.display = 'none';
      return true;
    }

      const errMsg = document.getElementById('confirmErrorMsg');
      errMsg.style.display = 'none';
      errMsg.innerText = '';
      btn.disabled = true;
      btn.innerText = 'กำลังบันทึก...';

      _pendingFd.clientIp = await getClientIp();

      google.script.run
        .withSuccessHandler(res => {
          btn.disabled = false;
          btn.innerText = '✅ ยืนยันการจอง';
          if (res.success) {
            document.getElementById('confirmPage').style.display = 'none';
            document.getElementById('mainContent').style.display = '';
            delete dataCache[_pendingFd.bookingDate];
            loadDate(_pendingFd.bookingDate);
            document.getElementById('successCode').innerText = formatCode(res.code);
            document.getElementById('successDetail').innerHTML = `ห้อง <strong>${_pendingFd.roomName}</strong><br>${_pendingFd.startTime} – ${_pendingFd.endTime} น.<br>${_pendingFd.bookingDate.split('-').reverse().join('/')}`;
            document.getElementById('successOverlay').classList.add('show');
            _pendingFd = null;
          } else {
            // ✅ FIX: แทนที่ alert() ด้วยการแสดง error message สีแดงบนหน้า
            errMsg.innerText = '❌ ' + (res.message || 'เกิดข้อผิดพลาด กรุณาลองใหม่');
            errMsg.style.display = 'block';
            errMsg.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
          }
        })
        .withFailureHandler(() => {
          btn.disabled = false;
          btn.innerText = '✅ ยืนยันการจอง';
          // ✅ FIX: แทนที่ alert() ด้วยการแสดง error message สีแดงบนหน้า
          errMsg.innerText = '❌ เกิดข้อผิดพลาด กรุณาลองใหม่';
          errMsg.style.display = 'block';
          errMsg.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
        })
        .saveBooking(_pendingFd);
    }

    // ===== SEARCH =====
    function searchBooking() {
      const id = document.getElementById('searchId').value.trim();
      // แก้ไข RegExp literal จาก double backslash ให้เป็น single backslash เพื่อให้ตรวจรหัสนิสิต 10 หลักได้ถูกต้อง
      if (!/^\d{10}$/.test(id)) { const errDiv = document.getElementById('searchResult'); errDiv.innerHTML = '<div style="font-size:0.8rem;color:#dc3545;padding:8px 0;">กรุณากรอกรหัสนิสิต 10 หลัก</div>'; return; }
      const resDiv = document.getElementById('searchResult');
      resDiv.innerHTML = '<div style="font-size:0.8rem;color:#999;padding:8px 0;">กำลังค้นหา...</div>';

      google.script.run
        .withSuccessHandler(data => {
          // ✅ FIX: เดิมกรองผลลัพธ์ให้เหลือแค่วันที่กำลังเปิดดูอยู่บนตารางเท่านั้น (currentDate)
          // ทำให้ถ้านิสิตมีการจองที่ยังใช้งานอยู่จริงแต่เป็นวันอื่น (เช่น จองไว้พรุ่งนี้ แต่กำลังเปิดดูวันนี้)
          // จะขึ้น "ไม่พบข้อมูลการจอง" ทั้งที่จริงมีการจองอยู่ — เอาการกรองตามวันที่ออก แสดงทุกรายการที่ยัง active ทั้งหมด
          const filtered = data || [];
          if (!filtered.length) { resDiv.innerHTML = '<div style="font-size:0.8rem;color:#999;padding:8px 0;">ไม่พบข้อมูลการจอง</div>'; return; }
          const statusClass = s => s === 'มาแล้ว' || s === 'เช็คอินแล้ว' ? 'status-checkedin' : s === 'รอเช็คอิน' ? 'status-waiting' : 'status-other';
          const statusLabel = s => s === 'มาแล้ว' || s === 'เช็คอินแล้ว' ? 'เช็คอินแล้ว' : s === 'รอเช็คอิน' ? 'รอเช็คอิน' : s;
          resDiv.innerHTML = filtered.map(b => `
        <div class="booking-result-card">
          <div class="room">${b.room} <span class="status-badge ${statusClass(b.status)}">${statusLabel(b.status)}</span></div>
          <div class="meta">${b.start} – ${b.end} · ${b.date}</div>
          <div class="meta">Code: <strong>${formatCode(b.code)}</strong></div>
        </div>`).join('');
        })
        .withFailureHandler(() => { resDiv.innerHTML = '<div style="color:#dc3545;font-size:0.8rem;">โหลดไม่สำเร็จ</div>'; })
        .getMyBookings(id);
    }
